# src/features.py
def engineer_features(df):
    """Generates 22 new features from raw credit data"""
    
    # 1. Payment Delays
    df['MAX_DELAY'] = df[['PAY_0'...'PAY_6']].max(axis=1)
    
    # 2. Utilization Ratios
    df['UTILIZATION'] = df['BILL_AMT1'] / df['LIMIT_BAL']
    
    # 3. Interaction Terms
    df['RISK_SCORE'] = df['MAX_DELAY'] * df['UTILIZATION']
    
    return scale(df) # StandardScaler
